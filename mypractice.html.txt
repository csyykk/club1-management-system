<script>
checkLogin();
function loadMyRecord(){
    let userData = getUserData();
    let actList = userData.activityList || [];
    let tbody = document.getElementById("practiceBody");
    tbody.innerHTML = "";
    actList.forEach(item=>{
        if(item.title && item.score){
            tbody.innerHTML += `
            <tr>
                <td>${item.title}</td>
                <td>${item.time}</td>
                <td class="score-cell">${item.score}</td>
                <td>${item.status=="recruit"?"进行中":"已结束"}</td>
            </tr>
            `
        }
    })
}
window.onload = function(){
    loadMyRecord();
}
